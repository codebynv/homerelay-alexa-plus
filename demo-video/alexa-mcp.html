export const CheckPantryIntentHandler = {
  async handle(handlerInput) {
    const ingredients = getSlots(handlerInput);
    // Connect to MCP Server via Streamable HTTP
    await client.connect(transport);
    const res = await client.callTool({
      name: 'check_pantry',
      arguments: { ingredients }
    });
    const speech = `Available: ${res.available}. Missing: ${res.missing}.`;
    return responseBuilder.speak(speech).getResponse();
  }
};
